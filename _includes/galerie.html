{%- comment -%} Bildergalerie für Neuigkeiten. Ohne Parameter (automatisch unter jedem Post): zeigt alle Bilder direkt im Ordner assets/images/neuigkeiten// Mit Parameter, z. B. mitten im Text unter einem Zwischentitel: {% include galerie.html ordner="offiziere" %} zeigt die Bilder im Unterordner ...//offiziere/ Mit Parameter "pfad" (z. B. für die Seite /galerie/): {% include galerie.html pfad="/assets/images/galerie/2025-hauptuebung/" alt="Hauptübung 2025" %} zeigt die Bilder in genau diesem Ordner; "alt" ersetzt den Seitentitel in der Bildbeschreibung. Sortierung alphabetisch nach Dateiname. {%- endcomment -%} {%- if include.pfad -%} {%- assign ordner = include.pfad -%} {%- assign letztes = ordner | slice: -1 -%} {%- if letztes != '/' -%}{%- assign ordner = ordner | append: '/' -%}{%- endif -%} {%- else -%} {%- assign post_name = page.path | split: '/' | last | remove: '.md' -%} {%- assign ordner = '/assets/images/neuigkeiten/' | append: post_name | append: '/' -%} {%- if include.ordner -%} {%- assign ordner = ordner | append: include.ordner | append: '/' -%} {%- endif -%} {%- endif -%} {%- assign alt_text = include.alt | default: page.title -%} {%- assign bilder = site.static_files | where_exp: "f", "f.path contains ordner" | sort: 'path' -%} {%- assign anzahl = 0 -%} {%- capture items -%} {%- for f in bilder -%} {%- assign rest = f.path | remove_first: ordner -%} {%- if rest contains '/' -%}{%- continue -%}{%- endif -%} {%- assign ext = f.extname | downcase -%} {%- if ext == '.jpg' or ext == '.jpeg' or ext == '.png' or ext == '.webp' or ext == '.gif' -%} {%- assign anzahl = anzahl | plus: 1 -%} {%- endif -%} {%- endfor -%} {%- endcapture -%} {%- if anzahl > 0 -%}
{{ items }}
